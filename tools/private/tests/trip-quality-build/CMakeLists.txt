cmake_minimum_required(VERSION 3.22)
project(navi_trip_quality_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
get_filename_component(NAVI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
include("${NAVI_ROOT}/cmake/NaviValidation.cmake")
set(ANALYSIS_SOURCE "${NAVI_ROOT}/libs/map/navi_trip_analysis.cpp")
add_library(trip_quality STATIC "${NAVI_ROOT}/libs/geometry/distance_on_sphere.cpp")
if(EXISTS "${ANALYSIS_SOURCE}")
  target_sources(trip_quality PRIVATE "${ANALYSIS_SOURCE}")
endif()
file(GLOB BOOST_HEADERS "${NAVI_ROOT}/3party/boost/libs/*/include")
target_include_directories(trip_quality PUBLIC "${NAVI_ROOT}" "${NAVI_ROOT}/libs" ${BOOST_HEADERS})
target_compile_options(trip_quality PRIVATE -O2 -ffast-math)
navi_require_finite_math("${ANALYSIS_SOURCE}")
add_executable(trip_quality_probe ../trip_quality_probe.cpp)
target_link_libraries(trip_quality_probe PRIVATE trip_quality)
