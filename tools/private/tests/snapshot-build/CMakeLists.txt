cmake_minimum_required(VERSION 3.22)
project(navi_snapshot_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
get_filename_component(NAVI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
include("${NAVI_ROOT}/cmake/NaviValidation.cmake")
set(NAVI_BOUNDARY "${NAVI_ROOT}/libs/map/navi_route_snapshot.cpp")
add_library(snapshot_boundary STATIC "${NAVI_BOUNDARY}")
target_include_directories(snapshot_boundary PUBLIC "${NAVI_ROOT}/libs")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(snapshot_boundary PRIVATE -O2 -ffast-math)
endif()
navi_require_finite_math("${NAVI_BOUNDARY}")
add_executable(snapshot_tests ../navi_snapshot_test.cpp)
target_link_libraries(snapshot_tests PRIVATE snapshot_boundary)
if(MINGW)
  target_link_options(snapshot_tests PRIVATE -static)
endif()
